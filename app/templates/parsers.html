Minimal plugin

from app.parsers.base import Entity, ParseResult, Parser

class ExampleParser(Parser):
    name = "example_scanner"
    description = "Example scanner JSONL"
    parser_version = "1"

    def detect(self, text: str) -> float:
        return 0.9 if '"example_scan"' in text else 0.0

    def parse_detailed(self, text: str) -> ParseResult:
        result = ParseResult(parser_version=self.parser_version)
        for line in text.splitlines():
            if not line.strip():
                continue
            result.entities.append(Entity(type="domain", value=line.strip()))
        return result

PARSER = ExampleParser()
